วิธีตั้งค่า Apps Script ของคุณเอง (ทำครั้งเดียว, 5 นาที) ▾
  1. เปิด script.google.com/create (ล็อกอินด้วยบัญชีเดียวกับ Drive)
  2. ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดด้านล่างนี้แทน
/**
 * SBK Traffic Monitoring System — Google Drive Bridge
 * ============================================================
 * ใช้ JSONP (ผ่าน <script> tag) แทน fetch() ธรรมดา — ไม่ติดปัญหา CORS
 * เลย ไม่ว่าเปิดไฟล์ HTML แบบ file:// (ดับเบิลคลิก) หรือจากเว็บเซิร์ฟเวอร์
 * ก็ตาม เพราะ <script src="..."> ไม่ถูกจำกัดโดย Same-Origin Policy
 *
 * โหมดการใช้งาน (ส่งผ่าน query string):
 *   ?mode=list                       → รายชื่อไฟล์ + เวลาแก้ไขล่าสุด (เบา เร็ว)
 *   ?mode=fetch&names=A|B            → ข้อมูลเฉพาะไฟล์ที่ระบุ (คั่นด้วย |)
 *   ?mode=fetch                      → ข้อมูลทุกไฟล์ในโฟลเดอร์
 *   เพิ่ม &callback=fnName ต่อท้ายได้เสมอ → ตอบกลับเป็น JSONP (fnName(...))
 *   ไม่ใส่ callback → ตอบกลับเป็น JSON ปกติ (เข้ากันได้กับโค้ดเวอร์ชันก่อนหน้า)
 *
 * ถ้าไม่ส่ง folderId มา จะใช้ DEFAULT_FOLDER_ID ด้านล่างอัตโนมัติ
 * ============================================================
 */

var DEFAULT_FOLDER_ID = '1YJZGlGjGbwn8-A9MChc--yCVSSdl28nu';

function doGet(e) {
  var params   = (e && e.parameter) ? e.parameter : {};
  var folderId = params.folderId || DEFAULT_FOLDER_ID;
  var mode     = params.mode || 'fetch';
  var only     = params.names ? String(params.names).split('|') : null;
  var callback = params.callback || null;   // ★ ชื่อฟังก์ชัน JSONP ที่ฝั่งเว็บขอมา

  try {
    var folder = DriveApp.getFolderById(folderId);

    if (mode === 'list') {
      var list = [];
      var it1 = folder.getFiles();
      while (it1.hasNext()) {
        var f1 = it1.next();
        if (!isSpreadsheet_(f1)) continue;
        list.push({ name: f1.getName(), updated: f1.getLastUpdated().getTime(), size: f1.getSize() });
      }
      return respond_({ mode: 'list', folderId: folderId, files: list }, callback);
    }

    var out = [];
    var it2 = folder.getFiles();
    while (it2.hasNext()) {
      var f2 = it2.next();
      if (!isSpreadsheet_(f2)) continue;

      var nm = f2.getName();
      if (only && only.indexOf(nm) === -1) continue;

      try {
        if (f2.getMimeType() === MimeType.GOOGLE_SHEETS) {
          var ss = SpreadsheetApp.openById(f2.getId());
          var sheets = ss.getSheets().map(function (sh) {
            return { name: sh.getName(), rows: sh.getDataRange().getValues() };
          });
          out.push({ name: nm, updated: f2.getLastUpdated().getTime(), sheets: sheets });
        } else {
          out.push({
            name: nm,
            updated: f2.getLastUpdated().getTime(),
            base64: Utilities.base64Encode(f2.getBlob().getBytes())
          });
        }
      } catch (fileErr) {
        out.push({ name: nm, error: String(fileErr) });
      }
    }
    return respond_({ mode: 'fetch', folderId: folderId, files: out }, callback);

  } catch (err) {
    return respond_({ error: String(err) }, callback);
  }
}

/** ตรวจว่าเป็นไฟล์สเปรดชีตที่อ่านได้หรือไม่ */
function isSpreadsheet_(file) {
  var mime = file.getMimeType();
  if (mime === MimeType.GOOGLE_SHEETS) return true;
  if (mime === MimeType.MICROSOFT_EXCEL) return true;
  return /\.(xlsx|xls)$/i.test(file.getName());
}

/** ส่งผลลัพธ์กลับ — เป็น JSONP ถ้ามี callback มา (ข้ามปัญหา CORS ทั้งหมด),
 *  หรือ JSON ปกติถ้าไม่มี (สำหรับเรียกจาก server อื่นที่ไม่ติด CORS อยู่แล้ว) */
function respond_(obj, callback) {
  var json = JSON.stringify(obj);
  if (callback) {
    // escape ชื่อ callback ให้ปลอดภัย (อนุญาตเฉพาะตัวอักษร ตัวเลข _ $)
    var safeCb = String(callback).replace(/[^\w$]/g, '');
    return ContentService.createTextOutput(safeCb + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

/**
 * ★ รันฟังก์ชันนี้ 1 ครั้งจากปุ่ม Run ▶ ก่อน Deploy ★
 * เว็บแอปที่ deploy แล้วจะเด้งขอสิทธิ์เองไม่ได้ ต้องอนุญาตล่วงหน้าจากหน้าแก้ไข
 */
function authorizeScopes() {
  DriveApp.getRootFolder();
  var f = DriveApp.getFolderById(DEFAULT_FOLDER_ID);
  var it = f.getFiles();
  if (it.hasNext()) {
    var first = it.next();
    if (first.getMimeType() === MimeType.GOOGLE_SHEETS) {
      SpreadsheetApp.openById(first.getId()).getSheets().length;
    }
  }
  Logger.log('อนุญาตสิทธิ์เรียบร้อย — โฟลเดอร์: ' + f.getName());
}
  1. เลือกฟังก์ชัน authorizeScopes จาก dropdown ข้างปุ่ม Run ▶ (ด้านบนหน้าต่างโค้ด) แล้วกด Run
  2. จะมีหน้าต่างขอสิทธิ์ขึ้นมา → Review permissions → เลือกบัญชี → หากขึ้นเตือน "Google hasn't verified this app" กด Advanced → Go to (ชื่อโปรเจกต์) (unsafe) → Allow
  3. กด Deploy → New deployment → เลือกประเภท Web app
  4. ตั้งค่า Execute as: Me และ Who has access: Anyone แล้วกด Deploy
  5. คัดลอก Web app URL (ลงท้ายด้วย /exec) มาวางในช่อง "ตั้งค่าขั้นสูง" ด้านบน

🔑 ขั้นตอนที่ 3 (รัน authorizeScopes) สำคัญมาก — Web App ที่ deploy แล้วจะเด้งขอสิทธิ์เองไม่ได้ ต้องรันจากหน้าแก้ไข 1 ครั้งเพื่ออนุญาตสิทธิ์ Drive/Sheets ก่อนเสมอ

⚠️ URL นี้จะเปิดให้ใครก็ตามที่มีลิงก์อ่านไฟล์ในโฟลเดอร์นี้ได้ (ไม่ต้องล็อกอิน) — อย่าเผยแพร่ URL นี้แบบสาธารณะ หากต้องการจำกัดเพิ่มเติม เลือก "Anyone with Google account" แทนตอน Deploy
