ดึงข้อมูลจาก Google Drive

อ่านไฟล์ Excel/Sheets ทุกไฟล์ในโฟลเดอร์มาวิเคราะห์

✅ ระบบตั้งค่าไว้ให้แล้ว — ทั้ง Web App URL และ Folder ID ถูกฝังอยู่ในโปรแกรม เปิดไฟล์แล้วดึงข้อมูลเองอัตโนมัติ ช่องด้านบนมีไว้เผื่อต้องการเปลี่ยนโฟลเดอร์/สคริปต์ในอนาคตเท่านั้น

วิธีอัปเดตโค้ด Apps Script ให้รองรับซิงก์อัตโนมัติ (ทำครั้งเดียว) ▾
  1. เปิด script.google.com/create (ล็อกอินด้วยบัญชีเดียวกับ Drive)
  2. ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดด้านล่างนี้แทน
/** SBK Traffic — Drive Bridge v32
 *  รองรับ 3 โหมดในสคริปต์เดียว:
 *    ?folderId=XXX              -> ส่งไฟล์ทั้งหมด (แบบเดิม)
 *    ?folderId=XXX&mode=list     -> ส่งเฉพาะรายชื่อ+เวลาแก้ไข (เบามาก ใช้เช็คว่ามีอะไรใหม่)
 *    ?folderId=XXX&ids=a,b,c     -> ส่งเฉพาะไฟล์ที่ระบุ (ดึงเฉพาะไฟล์ที่เปลี่ยน)
 */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var folderId = p.folderId;
  if (!folderId) return respond({ error: 'ไม่พบ folderId' });
  try {
    var folder   = DriveApp.getFolderById(folderId);
    var listOnly = (p.mode === 'list');
    var wanted   = p.ids ? String(p.ids).split(',') : null;
    var it = folder.getFiles();
    var out = [];
    while (it.hasNext()) {
      var f = it.next();
      var name = f.getName();
      var mime = f.getMimeType();
      var isSheet = (mime === MimeType.GOOGLE_SHEETS);
      var isExcel = (mime === MimeType.MICROSOFT_EXCEL || /\.(xlsx|xls)$/i.test(name));
      if (!isSheet && !isExcel) continue;              // ข้ามไฟล์ประเภทอื่น

      var id = f.getId();
      var meta = { id: id, name: name, modified: f.getLastUpdated().getTime() };
      if (listOnly) { out.push(meta); continue; }        // โหมดสำรวจ: ส่งแค่ metadata
      if (wanted && wanted.indexOf(id) === -1) continue; // ขอเฉพาะไฟล์ที่ระบุ

      try {
        if (isSheet) {
          // Google Sheets: อ่านค่าตรง ๆ ด้วย SpreadsheetApp (ไม่ต้องแปลงไฟล์ ไม่ใช้ UrlFetchApp)
          var ss = SpreadsheetApp.openById(id);
          meta.sheets = ss.getSheets().map(function (sh) {
            return { name: sh.getName(), rows: sh.getDataRange().getValues() };
          });
        } else {
          // ไฟล์ .xlsx แท้: ส่ง bytes เป็น base64 ให้ฝั่งเว็บ parse ด้วย SheetJS
          meta.base64 = Utilities.base64Encode(f.getBlob().getBytes());
        }
      } catch (fileErr) {
        meta.error = String(fileErr);                    // ไฟล์เดียวพัง ไม่ทำให้ทั้งชุดล้ม
      }
      out.push(meta);
    }
    return respond({ mode: listOnly ? 'list' : 'files', files: out });
  } catch (err) {
    return respond({ error: String(err) });
  }
}

function respond(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// รันฟังก์ชันนี้ 1 ครั้งจากปุ่ม Run ▶ ในหน้าแก้ไข เพื่ออนุญาตสิทธิ์ล่วงหน้า
function authorizeScopes() {
  DriveApp.getRootFolder();
}
  1. เลือกฟังก์ชัน authorizeScopes จาก dropdown ข้างปุ่ม Run ▶ (ด้านบนหน้าต่างโค้ด) แล้วกด Run
  2. จะมีหน้าต่างขอสิทธิ์ขึ้นมา → Review permissions → เลือกบัญชี → หากขึ้นเตือน "Google hasn't verified this app" กด Advanced → Go to (ชื่อโปรเจกต์) (unsafe) → Allow
  3. กด Deploy → New deployment → เลือกประเภท Web app
  4. ตั้งค่า Execute as: Me และ Who has access: Anyone แล้วกด Deploy
  5. คัดลอก Web app URL (ลงท้ายด้วย /exec) มาวางในช่องด้านบน

🔑 ขั้นตอนที่ 3 (รัน authorizeScopes) สำคัญมาก — Web App ที่ deploy แล้วจะเด้งขอสิทธิ์เองไม่ได้ ต้องรันจากหน้าแก้ไข 1 ครั้งเพื่ออนุญาตสิทธิ์ Drive/Sheets ก่อนเสมอ · โค้ดเวอร์ชันนี้ (v27) ไม่ใช้ UrlFetchApp แล้ว จึงไม่ขอสิทธิ์ "เรียก URL ภายนอก" อีกต่อไป

🔧 ถ้าเคย deploy โค้ดเวอร์ชันเก่าแล้วเจอ error "UrlFetchApp ไม่ได้รับอนุญาต": ลบโค้ดเดิมทั้งหมด → วางโค้ดเวอร์ชันนี้แทน → รัน authorizeScopes ตามขั้นตอนด้านบน → แล้วไปที่ Deploy → Manage deployments → แก้ไข (ไอคอนดินสอ) → Version: New version → Deploy — ไม่ต้องสร้าง URL ใหม่

⚠️ URL นี้จะเปิดให้ใครก็ตามที่มีลิงก์อ่านไฟล์ในโฟลเดอร์นี้ได้ (ไม่ต้องล็อกอิน) — อย่าเผยแพร่ URL นี้แบบสาธารณะ หากต้องการจำกัดเพิ่มเติม เลือก "Anyone with Google account" แทนตอน Deploy
